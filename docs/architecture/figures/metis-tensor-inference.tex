      \pgfmathsetmacro{\ch}{#5/#6-.05}
      \pgfmathtruncatemacro{\idx}{mod((\rr-1)*#7+\cc-1,24)}
      \pgfmathtruncatemacro{\tone}{{55,30,80,55,80,30,55,55,30,80,80,55,80,55,30,55,30,80,55,30,80,55,55,80}[\idx]}
      \def\celltext{}

        \def\tone{55}\def\celltext{1}
        \ifnum\rr=1\ifnum\cc=#7\def\tone{0}\def\celltext{0}\fi\fi
      \fi
      \ifnum#9=2
        \def\tone{30}\pgfmathtruncatemacro{\pn}{\cc-1}\edef\celltext{\pn}
      \fi
      \ifnum#9=3
        \def\tone{30}\def\celltext{$\tau$}
        \ifnum\rr=1\ifnum\cc=#7\def\tone{0}\def\celltext{}\fi\fi
      \fi
      \path[fill=#8!\tone,rounded corners=.6pt]
        (\xx,\yy) rectangle +(\cw,\ch);
      \ifnum#9=2
        \draw[black!15,line width=.3pt,rounded corners=.6pt]
          (\xx,\yy) rectangle +(\cw,\ch);
      \fi
      \node[font=\scriptsize,inner sep=0pt,text=ink]
        at (\xx+\cw/2,\yy+\ch/2) {\celltext};
    }}
    \draw[black!55,line width=.55pt]
      (-#4/2+.03,#5/2+.08)--(-#4/2-.08,#5/2+.08)
      --(-#4/2-.08,-#5/2-.08)--(-#4/2+.03,-#5/2-.08);

      (#4/2-.03,#5/2+.08)--(#4/2+.08,#5/2+.08)
